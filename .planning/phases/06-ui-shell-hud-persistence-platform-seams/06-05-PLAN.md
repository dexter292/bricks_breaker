// Preload on PlayingHost mount / Play (research lock — yes):
const previousBestRef = useRef(0);
useEffect(() => {
  void createAsyncStoragePersonalBestStore()
    .getBest()
    .then((b) => { previousBestRef.current = b; })
    .catch(() => { previousBestRef.current = 0; });
}, []);

// Cold path — NEVER await inside useAnimatedReaction / frame callback:
function handleRunEnded(runScore: number, outcome: 'win' | 'lose') {
  const previous = previousBestRef.current;
  const { best, isNewRecord } = evaluatePersonalBest(runScore, previous);
  setResultBest(best);
  setIsNewRecord(isNewRecord);
  if (isNewRecord) {
    previousBestRef.current = best;
    void store.setBest(best).catch(() => {});
  }
  const payload = { score: runScore, outcome, isNewRecord };
  platform.ads.onRunEnded(payload);
  platform.purchases.onRunEnded(payload);
  platform.accounts.onRunEnded(payload);
}

// ResultOverlay props:
