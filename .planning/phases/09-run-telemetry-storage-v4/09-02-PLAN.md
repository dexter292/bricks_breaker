Bump `ProgressBlob` from v3 to v4: add the `telemetry` sub-object (lifetime aggregate + per-
`(mode, levelId)` aggregate + a bounded recent-run ring), extend the fail-soft parse contract so
telemetry corruption cannot wipe progress, and extend the migrate chain one link deeper — all
while renaming (not rewriting) the current v3-active functions so the existing, already-tested
v3/v2/v1 path is reused verbatim.

Purpose: N-STAT-02's lossless migration + fail-soft contract, and the schema N-STAT-01's runtime
counters persist into.
Output: `types.ts`/`parseBlob.ts`/`migrateProgress.ts`/`watermark.ts` extended, `telemetry.ts`
(new pure merge helpers), barrel updated, existing v3 test file's renamed references updated.

export type GameMode = 'campaign' | 'endless' | 'daily';
export type RunOutcome = 'win' | 'lose' | 'abandoned';
export const RECENT_RUNS_BOUND = 50 as const; // ~120 bytes/entry x 50 ~= 6KB, vs ~2MB Android CursorWindow practical ceiling (A3) — orders of magnitude headroom; "a round number in the tens" per CONTEXT discretion.

export type TelemetryAggregate = {
  runsPlayed: number; runsWon: number; runsLost: number; runsAbandoned: number;
  bricksBroken: number; bestComboEver: number;
  pickupMultiball: number; pickupExpand: number; pickupExtraLife: number; pickupSlow: number; pickupFireball: number;
  livesLost: number; longestRallyEver: number; largestCascadeEver: number;
  ticksPlayed: number; wallClockMsTotal: number;
};
export type RunLogEntry = { mode: GameMode; levelId: string; outcome: RunOutcome; score: number; ticks: number; timestamp: number };
export type RunStatsInput = {
  bricksBroken: number; bestCombo: number;
  pickupMultiball: number; pickupExpand: number; pickupExtraLife: number; pickupSlow: number; pickupFireball: number;
  livesLost: number; longestRally: number; largestCascade: number;
  ticksPlayed: number; wallClockMs: number;
}; // NOTE: intentionally excludes `rallyCurrent` (runtime-internal bookkeeping only, see runStats.ts) and never imports src/runtime/* (LC boundary: services may import core/services only, never runtime)
export type TelemetryBlob = {
  lifetime: TelemetryAggregate;
  byMode: { campaign: Partial<Record<string, TelemetryAggregate>>; endless: Partial<Record<string, TelemetryAggregate>>; daily: Partial<Record<string, TelemetryAggregate>> };
  recentRuns: RunLogEntry[];
};
