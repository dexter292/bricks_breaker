```typescript
export type PaddleGestureOptions = {
  /** SharedValue 0=docked,1=playing,2=won,3=lost — mirror SimPhase numeric */
  simPhase: SharedValue<number>;
  /** SharedValue ui shell: 0=playing,1=paused,2=countdown (define locally; document) */
  uiPhase: SharedValue<number>;
  camScale: SharedValue<number>;
  /** Initial / current absolute paddle center VU */
  initialPaddleX?: number;
};

export type PaddleGestureHandle = {
  paddleTarget: SharedValue<number>;
  launchFlag: SharedValue<number>;
  panActive: SharedValue<boolean>;
  gesture: GestureType; // composed Race
};
```

Implementation requirements (D-04, D-11, RESEARCH Pattern 1–2):
1. Create SharedValues: `paddleTarget` (default initialPaddleX ?? 180), `launchFlag` (0), `panActive` (false), plus worklet anchors `anchorPaddleX`.
2. `Gesture.Pan()`:
   - `.minDistance(PAN_MIN_DISTANCE_PX)`
   - `.enabled(...)` from React props derived by caller OR check ui/sim inside worklet: only when simPhase is DOCKED(0)|PLAYING(1) AND uiPhase is playing(0)
   - `onBegin`/`onStart`: `panActive=true`; `anchorPaddleX = paddleTarget` (**not** finger X)
   - `onUpdate`: `paddleTarget = computeRelativePaddleX({ anchorPaddleX, translationXPx: e.translationX, camScale: camScale.value, gain: PADDLE_GAIN, prevTarget: paddleTarget.value, smoothAlpha: SMOOTH_ALPHA, paddleHalfW: 36, logicalWidth: LOGICAL_WIDTH_VU })`
   - `onFinalize`: `panActive=false`
3. `Gesture.Tap()`:
   - `.maxDistance(TAP_MAX_DISTANCE_PX)`
   - enabled only when docked path allowed (simPhase===0 && uiPhase===0) — also gate in `onEnd`
   - `onEnd`: if `shouldAcceptServeTap({ simPhaseDocked: simPhase.value===0, uiPaused: uiPhase.value===1, countdown: uiPhase.value===2, panActive: panActive.value })` then `launchFlag.value = 1`
4. `gesture = Gesture.Race(pan, tap)`
5. **Never** `runOnJS` / `scheduleOnRN` in handlers (LC-07)
6. **Never** set paddle from `e.absoluteX` / `e.x` alone
7. Do not implement Resume on playfield
8. No `core` imports

Export `usePaddleGesture(options): PaddleGestureHandle`.

Document numeric uiPhase map in a short file comment matching Plan 04/05 (`0 playing, 1 paused, 2 countdown`).
